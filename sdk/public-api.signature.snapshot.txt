# Public API signature snapshot for @agentdid/sdk

## core/agent-card.d.ts
import { AgentSigner } from './signer';
/** `uri` of the A2A extension through which a card names its agent DID and the keys it publishes. */
export declare const AGENT_DID_A2A_EXTENSION_URI = "https://agent-did.org/a2a/identity/v1";
/** The JWS `alg` of an Agent-DID card signature (RFC 8037). */
export declare const AGENT_CARD_SIGNATURE_ALG = "EdDSA";
/** Upper bound on `signatures` entries a verifier reads. */
export declare const MAX_AGENT_CARD_SIGNATURES = 16;
/** An entry of an Agent Card's `signatures` (A2A 1.0 `AgentCardSignature`). */
export interface AgentCardSignature {
    /** Base64url of the JWS protected header (a JSON object). */
    protected: string;
    /** Base64url of the signature over `protected + "." + base64url(JCS(card without signatures))`. */
    signature: string;
    /** Unprotected JWS header members. Not covered by the signature; the SDK does not read them. */
    header?: Record<string, unknown>;
}
/** An A2A Agent Card: any JSON object, optionally with `signatures`. */
export type AgentCard = Record<string, unknown> & {
    signatures?: AgentCardSignature[];
};
/** The parts of a card a verifier checks, once its format is right. */
export interface ParsedAgentCard {
    /** The agent DID every signature `kid` and the extension name. */
    did: string;
    /** One entry per signature, in order. */
    signatures: Array<{
        keyId: string;
        signingInput: string;
        signature: Uint8Array;
    }>;
    /** Keys the card publishes: signature `kid`s, then extension `keyIds`, without repeats. */
    publishedKeyIds: string[];
}
/** The DID of a DID URL with a non-empty fragment and nothing else (`did:method:id#fragment`), else `undefined`. */
export declare function didOfKeyId(keyId: unknown): string | undefined;
/**
 * Returns a copy of `card` with one more `signatures` entry, signed by `signer` as `keyId` (a DID URL of an
 * `assertionMethod` key of the agent). Signatures already on the card are kept and are not covered.
 */
export declare function signAgentCard(card: AgentCard, signer: AgentSigner, keyId: string): Promise<AgentCard>;
/**
 * Checks the format of a signed card without resolving anything: every `signatures` entry is an Agent-DID JWS
 * (protected header with `alg` `EdDSA`, a DID URL `kid`, `typ` `JOSE` when present, no `crit`; a 64-byte
 * signature), every `kid` and the extension name one DID. Returns `undefined` when the card does not have that
 * format (`format_drift`).
 */
export declare function parseSignedAgentCard(card: unknown): ParsedAgentCard | undefined;

     * card does not consume its nonce. Contract reasons are derived from the current DID document: a retired key is

     * verification methods. HTTP request signatures are verified with `verifyHttpRequestSignature`, not here.
     */
    verifySignature(did: string, payload: string, signature: string, keyId?: string, requiredPurpose?: VerificationRelationship): Promise<boolean>;
    /**
     * Verifies a signature made with `keyId` at the time the signed artifact states (RFC-001 §6.2.2), against the
     * DID's verified did:webvh log as `resolveDidWebvhState` gives it: the local log for a DID created or changed in
     * this client, otherwise the log the resolver fetches (the longer of it and a log imported with
     * `importDidWebvhHistory`), and the imported log only when the resolver does not fetch logs. A key retired by a
     * later entry of the log host is therefore seen even when an older log was imported.
     *
     * - `valid`: the signature verifies and the key was authorized for `requiredPurpose` at `statedTime`, with no
     *   later change that retires it, or with `trustedTimeNotAfter` earlier than that change.
     * - `invalid`: the signature does not verify, the key was not authorized at `statedTime`, the log cannot be
     *   fetched or fails verification (`did_resolution_failed`), or the controller binding does not hold now.
     * - `indeterminate`: the key was retired or the DID deactivated after `statedTime` and there is no trusted
     *   time evidence, or no did:webvh log is available for the DID. Do not accept it.
     *
     * The stated time is asserted by the signer, so a retired key is never accepted on the signer's word alone. With
     * `timestampToken` (an RFC 3161 token over the raw signature bytes) and `tsaTrustAnchors`, the SDK verifies the
     * token and uses its `genTime + accuracy` as the trusted time evidence (ADR-007); a token that does not verify gives
     * `invalid` / `timestamp_invalid`.
     */
    verifySignatureAtTime(did: string, payload: string, signature: string, keyId: string, options: VerifySignatureAtTimeOptions): Promise<VerifySignatureAtTimeResult>;
    /** The binding of `did` (and of its controllers) to its controller chain as it is now. */
    private assertDidWebvhBoundNow;
    /**
     * Issues an `AgentMetadataCredential` (RFC-001 §4.4): the agent's controller states the agent's metadata in a W3C
     * VC 2.0 credential secured with an `eddsa-jcs-2022` Data Integrity proof. `verificationMethodId` must be an active
     * Ed25519 `assertionMethod` key of the controller held by `signer`. The credential is presented out of band (an
     * A2A Agent Card, an MCP exchange, a request body); DID resolution never fetches it.
     */
    issueAgentMetadataCredential(params: IssueAgentMetadataCredentialParams): Promise<AgentMetadataCredential>;
    /**
     * Verifies an `AgentMetadataCredential` (RFC-001 §4.4): its form and validity period, that the agent DID resolves,
     * is not deactivated and is bound to its controller (RFC-001 §6.6 item 7), that `issuer` is that controller, and the
     * `eddsa-jcs-2022` proof by an active `assertionMethod` key of the issuer. Never throws for a bad credential: test
     * `result.ok`. The metadata in an accepted credential is the controller's assertion, not evidence of what runs.
     */
    verifyAgentMetadataCredential(credential: unknown, options?: VerifyAgentMetadataCredentialOptions): Promise<VerifyAgentMetadataCredentialResult>;
    /**
     * Delegates a capability (RFC-001 §6.7): grants `params.controller` the authority to act on `invocationTarget`

     * configuration); with it, the delegator must be the parent's `controller`, and the new capability may only narrow
     * the parent's target, actions and expiry.
     */
    createCapability(params: CreateCapabilityParams): Promise<AgentCapability>;
    /**
     * Verifies a delegated capability and its chain (RFC-001 §6.7): form, chain structure, the root controller the
     * verifier configured for the root target, attenuation at every link, validity period, and each proof by an active
     * Ed25519 `capabilityDelegation` key of its delegator, resolved now. Never throws for a bad capability: test
     * `result.ok`.
     */
    verifyCapabilityChain(capability: unknown, options: VerifyCapabilityChainOptions): Promise<VerifyCapabilityChainResult>;
    /** The delegation key `keyId` of `did`, resolved now, or why it cannot sign a delegation. */
    private delegationKeyCheck;
    /**
     * Verifies a request that invokes a capability (RFC-001 §6.5, §6.7): the request profile of
     * `verifyHttpRequestSignatureDetailed`, with the signing key under `capabilityInvocation` instead of
     * `assertionMethod`; the signature must cover `"agent-did-capability";key="<label>"`; the request's agent DID must
     * be the capability's `controller`, its URL within the target and its method allowed; then the capability chain
     * (`verifyCapabilityChain`). The nonce is recorded only when everything passes.
     */
    verifyHttpCapabilityInvocation(params: VerifyHttpCapabilityInvocationParams): Promise<VerifyHttpCapabilityInvocationResult>;
    /**
     * Verifies a JWT client assertion (`private_key_jwt`) against the agent DID registered for the client (RFC-001
     * §6.8): JWS form, `alg` (`EdDSA` or `Ed25519`), `typ` (absent or `client-authentication+jwt`), `kid` a DID URL of
     * `did`, `iss` = `sub` = `clientId`, `aud` exactly `issuer`, lifetime at most 300 s and times within the clock
     * skew, then the signature by an active Ed25519 `authentication` key of the resolved DID, and finally `jti` in the
     * nonce store. For an authorization server or MCP server that knows Agent-DID. Never throws: test `result.ok`.
     */
    verifyClientAssertion(params: VerifyClientAssertionParams): Promise<VerifyClientAssertionResult>;
    /**
     * Drops the resolver's cached resolution of `did`, so the next verification fetches and verifies it again (for a

    evictCachedResolution(did: string): void;

    /**

    resolve(did: string): Promise<AgentDIDDocument>;

     */
    resolveDocument(did: string): Promise<DidDocument>;
    /**
     * State of the locally stored did:webvh log: whether it ends in a deactivation entry and its
     * latest document. The log is verified (SCID, hash chain, controller proofs) before its state is
     * trusted; a log that fails verification is an error, never "not revoked". Returns undefined when
     * no log is stored for the DID.
     */
    private didWebvhState;
    resolveControllerChain(did: string, maxDepth?: number): Promise<DidDocument[]>;
    /**
     * Revokes a DID. For did:webvh DIDs this appends a controller-signed deactivation entry
     * (`deactivated: true`) to the DID's log, which is what makes the DID revoked; the
     * registry record is then updated as a derived index. Other DID methods are revoked in
     * the registry.
     */
    revokeDid(did: string, options?: DidWebvhLifecycleOptions): Promise<void>;
    private revokeDidLocked;
    updateDidDocument(did: string, patch: UpdateAgentDocumentParams, options?: DidWebvhLifecycleOptions): Promise<AgentDIDDocument>;
    private updateDidDocumentLocked;
    rotateVerificationMethod(did: string, options?: RotateVerificationMethodOptions): Promise<RotateVerificationMethodResult>;
    private rotateVerificationMethodLocked;
    /**
     * Retires one key of `did` and leaves the others as they are (RFC-001 §6.6): the key gets a `deactivated` time and
     * leaves every verification relationship. A planned retirement ends the overlap of a rotation started with
     * `addVerificationMethod`; signatures under the key are then rejected (`rotation_window_closed` in the composition
     * contract). With `emergency` the key also gets `revoked` and is never accepted again, whatever time a signature
     * states (`emergency_revoked`); an already retired key can still be revoked this way.
     *
     * Refuses an unknown key, a key already retired (without `emergency`) or revoked, and the last `authentication` or
     * `assertionMethod` key: use `rotateVerificationMethod` or `revokeDid` for those.
     */
    retireVerificationMethod(did: string, keyId: string, options?: RetireVerificationMethodOptions): Promise<AgentDIDDocument>;
    private retireVerificationMethodLocked;
    /**
     * Adds a key to the document of `did` without retiring any other key, listed under `options.relationships`
     * (default: `capabilityInvocation`). It is the first step of rotating the key that endorses the agents a DID
     * controls without unbinding them: add the new key, give each agent an entry signed by the new key, then retire
     * the old one with `rotateVerificationMethod` (RFC-001 §6.6 item 7).
     */
    addVerificationMethod(did: string, options?: AddVerificationMethodOptions): Promise<AddVerificationMethodResult>;
    private addVerificationMethodLocked;
    /**
     * Rotates the controller key of a did:webvh DID: appends a log entry whose `updateKeys` is
     * `newControllerSigner`'s key. This is the recovery path after a compromise of the controller key,
     * and it only works if the DID committed to a successor (pre-rotation: `webvh.nextUpdateKeys` at
     * creation, or a previous rotation that set `nextUpdateKeys`).
     *
     * - Pre-rotation active: `newControllerSigner` must be a committed key and signs the entry; the key
     *   that was active can no longer sign.
     * - Not active: the current controller signer (`options.controllerSigner`, or the one retained in this
     *   process) signs, and `nextUpdateKeys` may start pre-rotation.
     *
     * Commit to the following key in `nextUpdateKeys`, or the key you rotate to stays the only valid next key.
     */
    rotateControllerKey(did: string, options: RotateControllerKeyOptions): Promise<DidWebvhControllerKeys>;
    private rotateControllerKeyLocked;
    /** The controller keys of a did:webvh DID, from its verified log (including pre-rotation commitments). */
    getDidWebvhControllerKeys(did: string): Promise<DidWebvhControllerKeys>;
    /**
     * Sets, changes or (with `null`) removes the witnesses of a did:webvh DID (ADR-007) by appending a log entry. The
     * witnesses in force approve that entry (through `AgentDidClientOptions.collectWitnessProofs`); the new rule
     * governs the entries after it. Returns the witness configuration now in force.
     */

## core/nonce-store.d.ts
/**
 * Replay protection for verified HTTP message signatures.
 *
 * `AgentIdentity.verifyHttpRequestSignature` records each accepted signature's
 * `nonce` (scoped to its `keyid`) until the signature's `expires` time and
 * rejects a second request carrying the same pair. Use a shared store (for
 * example Redis `SET key 1 NX EXAT expiresAt`) when several verifier instances
 * serve the same agents.
 *
 * Python parity: `agent_did_sdk.core.nonce_store`.
 */
/**
 * Grammar of an Agent-DID nonce (RFC-001 §6.5 item 3): 1 to 256 characters of `A-Z a-z 0-9 . _ ~ + / = -`. A nonce has
 * no space, so in a store key `"<keyid> <nonce>"` the text before the last space is always the `keyid`.
 */
export declare const NONCE_PATTERN: RegExp;
export interface NonceStore {
    /**
     * Records `key` until `expiresAt` (Unix seconds). Resolves `true` when the key
     * was not recorded yet, `false` when it was (a replay). Must be atomic.
     */
    checkAndStore(key: string, expiresAt: number): Promise<boolean>;
}
export interface InMemoryNonceStoreOptions {
    /**
     * Upper bound on live entries. When the store is full after purging expired
     * entries, new nonces are refused (fail closed) instead of evicting live ones,
     * which would reopen a replay window. Default: 100 000.
     */
    maxEntries?: number;
    /**
     * Upper bound on live entries of one scope: the part of the key before its last space, which is the `keyid` of an
     * HTTP signature (`"<keyid> <nonce>"`). It keeps one signer from filling the store and locking every other signer
     * out. Default: 10 000.
     */
    maxEntriesPerScope?: number;
    /** Clock in Unix seconds; injectable for tests. */
    now?: () => number;
}
/** Process-local `NonceStore`. Suitable for a single verifier instance. */
export declare class InMemoryNonceStore implements NonceStore {
    private readonly entries;
    private readonly scopeCounts;
    private readonly maxEntries;
    private readonly maxEntriesPerScope;

## core/safe-fetch.d.ts
/**
 * Default `fetch` for the SDK's HTTP sources.
 *
 * - Redirects are not followed: targets are validated before the request is sent, so a redirect
 *   would move it to a destination that was never validated. A 3xx surfaces as a non-OK status.
 * - Host names are resolved once, at connection time, and the connection is refused when any
 *   resolved address is not globally routable. The address that was validated is the address that
 *   is connected to, so a DNS answer that changes after the check (rebinding) cannot redirect the
 *   request. TLS still verifies the original host name. Each request opens its own connection, never a pooled
 *   one, so a socket another part of the process opened to the same host is not reused unchecked.
 *
 * - Every request has one deadline (`timeoutMs`) covering lookup, connection and the whole body,
 *   and a maximum body size (`maxResponseBytes`). It asks for an unencoded body and rejects any
 *   other `Content-Encoding`, so the size limit applies to the bytes that are parsed.
 *
 * IP literals and `allowPrivateTargets` use the platform `fetch` (with redirects disabled and the
 * same limits), since there is nothing to resolve or the caller opted out. Callers that supply
 * their own `fetchFn` own their redirect, DNS and limit policy. Node.js only: it relies on
 * `node:http`, `node:https` and `node:dns`.
 */
import { HttpSecurityOptions, HttpTargetValidationOptions } from './http-security';
type AnyInit = Record<string, unknown> | undefined;
type LookupFn = (hostname: string) => Promise<Array<{
    address: string;
}>>;
export interface SafeFetchResponse {
    ok: boolean;
    status: number;
    headers: Headers;
    text: () => Promise<string>;
    json: () => Promise<unknown>;
}
/** Resolves the host of `url` and throws unless every address is globally routable. */
export declare function assertResolvedTargetAllowed(url: string, options?: HttpTargetValidationOptions, lookup?: LookupFn): Promise<void>;
export declare function createSafeFetch(options?: HttpSecurityOptions, lookup?: LookupFn, isBlockedAddress?: (address: string) => boolean): (url: string, init?: AnyInit) => Promise<any>;

/**
 * Why `verifyHttpRequestSignatureDetailed` rejected a request. When the request carries several
 * signatures, the reason is the one of the first signature that has `tag="agent-did"` (else the
 * first signature).
 *
 * - `missing_headers`: `Signature`, `Signature-Input` or `Agent-DID` is absent.
 * - `url_not_normalized`: the request URL is not in the normalized form of RFC-001 §6.5 item 4 (see
 *   `requestUrlFormProblem`); the SDK refuses to sign such a URL too.
 * - `malformed_headers`: `Signature` / `Signature-Input` / `Agent-DID` do not parse, no signature has a matching
 *   label, or the `Agent-DID` member of the signature's label is not a String without parameters.
 * - `digest_mismatch`: `Content-Digest` is absent or is not exactly the SHA-256 of the body.
 * - `tag_mismatch`: the `tag` parameter is not `agent-did` (for example another profile's signature).
 * - `alg_unsupported`: the `alg` parameter is not `ed25519`.
 * - `params_missing`: `created`, `expires`, `nonce` or `keyid` is absent or has the wrong type, or `nonce` is not 1 to
 *   256 characters of `A-Z a-z 0-9 . _ ~ + / = -` (RFC-001 §6.5 item 3).
 * - `keyid_not_bound`: `keyid` is not a key of the agent DID (the `Agent-DID` member of the signature's label).
 * - `components_missing`: a required covered component is absent.
 * - `components_order`: the required components are all covered but not first and in order.
 * - `created_skew`: `created` is further from now than `maxCreatedSkewSeconds`.
 * - `expires_before_created`: `expires` is earlier than `created`.
 * - `lifetime_exceeded`: `expires - created` is greater than `maxSignatureLifetimeSeconds`.
 * - `expired`: now is after `expires`.
 * - `signed_component_missing`: a covered component has no value in the request (for example no `Date`, or no
 *   `Agent-DID` member for the signature's label).
 * - `did_resolution_failed`: the agent DID has no active verification chain: unknown,
 *   revoked, deactivated, inactive controller, or the resolver failed. The SDK does not tell a
 *   resolver outage from a revocation; treat this reason as "could not establish the key".
 * - `keyid_not_authorized`: `keyid` is not an `assertionMethod` key of the DID document (a fragment
 *   the document does not have, or a key listed only under another relationship).
 * - `signature_invalid`: the signature does not verify, or the key is no longer active.
 * - `nonce_replayed`: the `(keyid, nonce)` pair was already seen.
 * - `did_not_allowed`: `allowedDids` is set and the agent DID is not in it.
 * - `controller_not_allowed`: `allowedControllers` is set and no DID of the controller chain is in it.
 */
export type HttpSignatureRejection = 'missing_headers' | 'url_not_normalized' | 'malformed_headers' | 'digest_mismatch' | 'tag_mismatch' | 'alg_unsupported' | 'params_missing' | 'keyid_not_bound' | 'components_missing' | 'components_order' | 'created_skew' | 'expires_before_created' | 'lifetime_exceeded' | 'expired' | 'signed_component_missing' | 'did_resolution_failed' | 'controller_not_bound' | 'keyid_not_authorized' | 'signature_invalid' | 'nonce_replayed' | 'did_not_allowed' | 'controller_not_allowed';

/**
 * Why a key was not authorized at the stated time (RFC-001 §6.2.2):
 * - `invalid_stated_time`: the stated time is not an RFC 3339 date-time.
 * - `did_not_yet_created`: the stated time is before the first entry of the DID log.
 * - `did_deactivated`: the DID was already deactivated at the stated time.
 * - `key_not_found`: the DID document at the stated time has no such key (or it has no public key).
 * - `key_deactivated`: the key was already deactivated at the stated time.
 * - `key_purpose_violation`: the key was not listed under the required relationship at the stated time.
 * - `key_revoked`: some entry of the log marks the key `revoked` (emergency revocation): it is not authorized at
 *   any stated time, even with trusted time evidence.
 */
export type KeyAtTimeRejection = 'invalid_stated_time' | 'did_not_yet_created' | 'did_deactivated' | 'key_not_found' | 'key_deactivated' | 'key_purpose_violation' | 'key_revoked';
/**
 * Why a key check at a stated time could not be decided (RFC-001 §6.2.2). The stated time is asserted by the
 * signer, so it cannot place a signature on either side of a later change:
 * - `key_retired_after_stated_time`: the key was authorized at the stated time, but a later log entry retires
 *   it (removed, deactivated or moved out of the required relationship) and no trusted time evidence places
 *   the signature before that entry.
 * - `did_deactivated_after_stated_time`: same, for a later deactivation of the DID.
 * - `history_unavailable`: no verified did:webvh log is available for the DID, so its state at the stated
 *   time cannot be established.
 */
export type KeyAtTimeUncertainty = 'key_retired_after_stated_time' | 'did_deactivated_after_stated_time' | 'history_unavailable';
/**
 * Result of a key check at a stated time. `authorized` means the key was authorized at the stated time and,
 * either nothing later retires it (`basis: 'no_later_change'`), or trusted time evidence places the signature
 * before the first entry that does (`basis: 'time_evidence'`). Treat `indeterminate` as not accepted.
 */
export type KeyAtTimeResult = {
    status: 'authorized';
    keyId: string;
    publicKeyMultibase: string;
    /** The log version in effect at the stated time. */
    versionId: string;
    basis: 'no_later_change' | 'time_evidence';
} | {
    status: 'not_authorized';
    keyId: string;
    reason: KeyAtTimeRejection;
    /** Relationships listing the key at the stated time (empty when it was not there). */
    foundIn: VerificationRelationship[];
} | {
    status: 'indeterminate';
    keyId: string;
    reason: KeyAtTimeUncertainty;
    /** The key at the stated time, when it was authorized then. */
    publicKeyMultibase?: string;
    /** The first log entry that retires the key or deactivates the DID, when there is one. */
    boundaryVersionId?: string;
    /** When the key stopped being authorized (log `versionTime` or the key's `deactivated`). */

/**
 * A delegated capability (RFC-001 §6.7, ADR-008 D-12a), shaped as an Authorization Capability (ZCAP v0.4 data

/**

export type TimestampRejection =
/** Not a DER `TimeStampResp` or `TimeStampToken`, not exactly one signer, or no embedded signer certificate. */
'token_malformed'
/** The response status is not `granted` (0). */
 | 'not_granted'
/** The message imprint uses a hash other than SHA-256, SHA-384 or SHA-512. */
 | 'hash_algorithm_unsupported'
/** The message imprint is not the hash of the bytes the token should cover. */
 | 'imprint_mismatch'
/** The signer certificate lacks a critical extended key usage with `id-kp-timeStamping`. */
 | 'not_a_timestamping_certificate'

    /** The candidate log, ending with the entry to approve. It is not published yet. */
    log: string;
    /** The `did-witness.json` in force before this entry, if the log has one. */
    witnessFile?: string;
    /** The witness rule that governs the entry. */
    witness: DidWebvhWitnessConfig;
}
/**
 * Obtains the approvals of an entry from its witnesses (for example over the network, or in process with
 * `createDidWebvhWitnessProof`). Returning fewer approvals than the threshold makes the operation fail.
 */
export type DidWebvhWitnessCollector = (request: DidWebvhWitnessRequest) => Promise<DidWebvhWitnessProofEntry[]>;
/**
 * Resolution metadata exposed by Agent-DID for a did:webvh log. Watcher configuration is deliberately not
 * exposed (ADR-002, condition 7). Every value is a string, list or boolean, as DID Core §7.3 requires of DID
 * document metadata.
 */
export interface DidWebvhMetadata {
    versionId: string;
    versionTime: string;
    created: string;
    updated: string;
    scid: string;
    updateKeys: string[];

     * Pre-rotation commitments (`nextKeyHashes`) in force after this version: while non-empty, the next
     * entry must be signed by a key whose hash is listed here, and the current `updateKeys` can no longer
     * sign. Empty when pre-rotation is not active.
     */
    nextKeyHashes: string[];
    deactivated: boolean;
    /** Cache TTL in seconds; the did:webvh default (3600) when the log does not set one. */
    ttl: string;
    /** Witness threshold in force after this version (ADR-007); `"0"` when the DID has no witnesses. */
    witnessThreshold: string;
    /** Witness DIDs in force after this version; empty when the DID has no witnesses. */
    witnesses: string[];
}

    did: string;

    state: DidWebvhDocument;
    metadata: DidWebvhMetadata;
    /** Number of the resolved version: the integer prefix of `metadata.versionId`. */
    versionNumber: number;

    log: string;

     * The `did-witness.json` text that approves the log, when any entry is governed by a witness rule. Publish it
     * next to `did.jsonl`, and before it: a resolver rejects a witnessed entry without its approvals.
     */
    witnessFile?: string;
}

     * Public keys (Ed25519 multikeys) the next entry will be signed with. Only their hashes are written to the

    nextUpdateKeys?: string[];

    created?: string;

export interface UpdateDidWebvhLogParams {
    log: string;
    signer: AgentSigner;

    document?: Record<string, unknown>;

     * active they must all be committed in `nextKeyHashes`, and they default to the signer's key.
     */
    updateKeys?: string[];
    /**
     * Public keys (Ed25519 multikeys) for the pre-rotation commitment of the NEXT entry. Omitted: the
     * commitments in force stay as they are. An empty array ends pre-rotation. Must not include any of the
     * `updateKeys` this entry sets.
     */
    nextUpdateKeys?: string[];

    updated?: string;

     */
    witness?: DidWebvhWitnessConfig | null;

/**
 * Verifies a `did.jsonl` log (SCID, entry hash chain, `updateKeys` proofs,
 * version times) and returns the resolved document. Throws `DidWebvhError` on
 * any verification failure; it never returns a partially verified result.
 *
 * With `versionId`, `versionNumber` or `versionTime` it returns that version of the log. The whole log is
 * still verified, and `metadata.deactivated` describes the DID, not the selected version: it is true for
 * every version of a deactivated DID (DID Core section 7.1.2).
 */
export declare function resolveDidWebvhLog(logText: string, options?: ResolveDidWebvhLogOptions): Promise<DidWebvhLogResult>;
/**
 * Whether a did:webvh log has entries governed by a witness rule, so resolving it needs its `did-witness.json`.
 * Rule of did:webvh v1.0 (as `didwebvh-ts` applies it): an entry is governed by the rule in force before it, or,
 * when there is none, by the rule the entry itself sets. Throws `DidWebvhError` for a malformed log.
 */
export declare function didWebvhLogRequiresWitnesses(logText: string): boolean;

export declare function deriveDidWebvhWitnessUrl(did: string): string;

    /** The candidate log, ending with the entry to approve (`DidWebvhWitnessRequest.log`). */
    log: string;
    /** The `did-witness.json` in force before that entry (`DidWebvhWitnessRequest.witnessFile`). */
    witnessFile?: string;
    /** The witness's own key: its `did:key` must be in the rule that governs the entry. */
    signer: AgentSigner;
}
/**
 * A witness's approval of the last entry of a candidate log (did:webvh v1.0): an `eddsa-jcs-2022` proof over
 * `{ versionId }` made with the witness's `did:key`. The whole candidate log is verified first, including the
 * approvals of earlier entries, so a witness never approves an entry that does not extend a valid log.
 */
export declare function createDidWebvhWitnessProof(params: CreateDidWebvhWitnessProofParams): Promise<DidWebvhWitnessProofEntry>;
/**
 * The multikeys (`z6Mk...`) that made the proofs of the LAST entry of a log: the keys that endorsed the current
 * document. The log must already be verified (`resolveDidWebvhLog`), which guarantees each proof is valid and was
 * made by a key authorized by the `updateKeys` in force; this only reads which keys they were.
 */
export declare function didWebvhLastEntrySigners(logText: string): string[];
/** How a candidate log relates to a stored one (both already verified). */
export type DidWebvhLogRelation = 'equal' | 'extends' | 'prefix' | 'diverged';
/**
 * Compares two did:webvh logs entry by entry by `versionId` (each one hashes its entry and the previous
 * `versionId`, so equal ids mean equal prefixes). `extends`: the candidate holds every stored entry and
 * more. `prefix`: the candidate is a strict prefix of the stored log (older). `diverged`: they differ at
 * some entry. Callers verify both logs first; this only compares their shape.
 */
export declare function compareDidWebvhLogs(stored: string, candidate: string): DidWebvhLogRelation;
/**
 * Derives the HTTPS URL of a did:webvh DID's `did.jsonl` (did:webvh v1.0,
 * "The DID to HTTPS Transformation"). Applies the same host and path rules as
 * `createDidWebvhLog`: lowercase ASCII FQDN (no IP literals), optional port
 * encoded as `%3A`, unencoded path segments without `.`/`..`, and no path,
 * query or fragment in the DID. Throws `DidWebvhError('invalidDid')` otherwise.
 */
export declare function deriveDidWebvhLogUrl(did: string): string;
