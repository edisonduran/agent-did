    private readonly defaultWebvhDomain;
    private readonly clock;

     * log text it was computed from. It is recomputed whenever the stored log changes.
     */
    private readonly didWebvhStateCache;

    private readonly didWebvhSignerStore;

    constructor(options?: AgentDidClientOptions);
    private nowIsoTimestamp;
    /**
     * Creates a new Agent-DID Document (Passport) from raw parameters.
     * Automatically hashes sensitive IP (coreModel, systemPrompt) and generates the DID.

     * @param params The raw agent configuration (name, prompt, capabilities, etc.)

     * Canonical did:webvh v1.0 profile: the agent document is the genesis entry of a

     */
    private createDidWebvhAgent;
    private requireWebvhCreateOptions;
    /**
     * Returns the local bootstrap controller for a creator's `did:key`, creating it on first use.
     * Its did:webvh log is signed by a controller key generated with a CSPRNG; the key is
     * returned to the caller exactly once (`privateKey`) and never derived from public data.
     */
    private ensureBootstrapController;
    /**
     * Appends a controller-signed entry with `document` to the DID's did:webvh log and
     * returns the state as signed. The signer is `controllerSigner` when given, otherwise
     * the controller signer retained when the DID was created in this process.
     */
    private appendDidWebvhEntry;
    private requireDidWebvhLifecycle;
    private resolveActiveVerificationChain;
    private hasActiveVerificationMethod;
    /**
     * Verifies an Agent-DID RFC 9421 HTTP Message Signature.
     *
     * Requires the Agent-DID profile (covered components, `tag="agent-did"`, `alg="ed25519"`,
     * `created`, `expires`, `nonce`, a `keyid` under the `Signature-Agent` DID), a matching
     * `Content-Digest`, and a signature by an active `assertionMethod` key. The nonce is then
     * recorded in `nonceStore` (default: a process-wide `InMemoryNonceStore`), so a replayed
     * request is rejected. Returns `false` on any failure; use `verifyHttpRequestSignatureDetailed`
     * to learn why.
     */
    verifyHttpRequestSignature(params: VerifyHttpRequestSignatureParams): Promise<boolean>;
    /**
     * Same verification as `verifyHttpRequestSignature`, returning the reason of a rejection
     * (see `HttpSignatureRejection`). Test `result.ok`: the result object is truthy for a rejection too.
     */
    verifyHttpRequestSignatureDetailed(params: VerifyHttpRequestSignatureParams): Promise<VerifyHttpRequestSignatureResult>;
    /** `verifySignature` that tells apart the ways a verification can fail. */
    private checkSignature;
    /**
     * Verifies that a signature was produced by a specific Agent-DID.
     * Uses the configured resolver and registry to validate against active verification methods.
     */
    verifySignature(did: string, payload: string, signature: string, keyId?: string, requiredPurpose?: VerificationRelationship): Promise<boolean>;
    /**
     * Verifies a signature made with `keyId` at the time the signed artifact states (RFC-001 §6.2.2), using the
     * DID's did:webvh log as created or imported in this client (`importDidWebvhHistory`).
     *
     * - `valid`: the signature verifies and the key was authorized for `requiredPurpose` at `statedTime`, with no
     *   later change that retires it, or with `trustedTimeNotAfter` earlier than that change.
     * - `invalid`: the signature does not verify, the key was not authorized at `statedTime`, or the log fails
     *   verification.
     * - `indeterminate`: the key was retired or the DID deactivated after `statedTime` and there is no trusted
     *   time evidence, or this client has no did:webvh log for the DID. Do not accept it.

    /**

    rotateVerificationMethod(did: string, options?: DidWebvhLifecycleOptions): Promise<RotateVerificationMethodResult>;
    /**
     * Rotates the controller key of a did:webvh DID: appends a log entry whose `updateKeys` is
     * `newControllerSigner`'s key. This is the recovery path after a compromise of the controller key,
     * and it only works if the DID committed to a successor (pre-rotation: `webvh.nextUpdateKeys` at

    getDidWebvhControllerKeys(did: string): Promise<DidWebvhControllerKeys>;
    getDocumentHistory(did: string): AgentDocumentHistoryEntry[];
    /** Returns the DID's did:webvh v1.0 log (`did.jsonl`) as created or imported in this process. */
    exportDidWebvhHistory(did: string): string;
    /**
     * Verifies a did:webvh v1.0 log (SCID, hash chain, controller-signed proofs) and loads it
     * into this process: latest document, registry record and document history. Changing the
     * DID afterwards requires its controller signer (`options.controllerSigner`).

    importDidWebvhHistory(didLog: string): Promise<AgentDIDDocument>;
    saveDidWebvhHistoryToFile(did: string, filePath: string): Promise<void>;

    persistDidWebvhHistoryToSource(did: string, documentRef: string, source: DIDDocumentSource): Promise<void>;
    restoreDidWebvhHistoryFromSource(documentRef: string, source: DIDDocumentSource): Promise<AgentDIDDocument>;

    private appendHistory;

## core/nonce-store.d.ts
/**
 * Replay protection for verified HTTP message signatures.
 *
 * `AgentIdentity.verifyHttpRequestSignature` records each accepted signature's
 * `nonce` (scoped to its `keyid`) until the signature's `expires` time and
 * rejects a second request carrying the same pair. Use a shared store (for
 * example Redis `SET key 1 NX EXAT expiresAt`) when several verifier instances
 * serve the same agents.
 *
 * Python parity: `agent_did_sdk.core.nonce_store`.
 */
export interface NonceStore {
    /**
     * Records `key` until `expiresAt` (Unix seconds). Resolves `true` when the key
     * was not recorded yet, `false` when it was (a replay). Must be atomic.
     */
    checkAndStore(key: string, expiresAt: number): Promise<boolean>;
}
export interface InMemoryNonceStoreOptions {
    /**
     * Upper bound on live entries. When the store is full after purging expired
     * entries, new nonces are refused (fail closed) instead of evicting live ones,
     * which would reopen a replay window. Default: 100 000.
     */
    maxEntries?: number;
    /** Clock in Unix seconds; injectable for tests. */
    now?: () => number;
}
/** Process-local `NonceStore`. Suitable for a single verifier instance. */
export declare class InMemoryNonceStore implements NonceStore {
    private readonly entries;
    private readonly maxEntries;
    private readonly now;
    constructor(options?: InMemoryNonceStoreOptions);
    checkAndStore(key: string, expiresAt: number): Promise<boolean>;
    /** Number of live entries (expired ones may linger until the next purge). */
    get size(): number;
    private purge;
}

## core/safe-fetch.d.ts
/**
 * Default `fetch` for the SDK's HTTP sources.
 *
 * - Redirects are not followed: targets are validated before the request is sent, so a redirect
 *   would move it to a destination that was never validated. A 3xx surfaces as a non-OK status.
 * - Host names are resolved once, at connection time, and the connection is refused when any
 *   resolved address is not globally routable. The address that was validated is the address that
 *   is connected to, so a DNS answer that changes after the check (rebinding) cannot redirect the
 *   request. TLS still verifies the original host name.
 *
 * IP literals and `allowPrivateTargets` use the platform `fetch` (with redirects disabled), since
 * there is nothing to resolve or the caller opted out. Callers that supply their own `fetchFn`
 * own their redirect and DNS policy. Node.js only: it relies on `node:http`, `node:https` and
 * `node:dns`.
 */
import { HttpTargetValidationOptions } from './http-security';
type AnyInit = Record<string, unknown> | undefined;
type LookupFn = (hostname: string) => Promise<Array<{
    address: string;
}>>;
export interface SafeFetchResponse {
    ok: boolean;
    status: number;
    headers: Headers;
    text: () => Promise<string>;
    json: () => Promise<unknown>;
}
/** Resolves the host of `url` and throws unless every address is globally routable. */
export declare function assertResolvedTargetAllowed(url: string, options?: HttpTargetValidationOptions, lookup?: LookupFn): Promise<void>;
export declare function createSafeFetch(options?: HttpTargetValidationOptions, lookup?: LookupFn, isBlockedAddress?: (address: string) => boolean): (url: string, init?: AnyInit) => Promise<any>;
export {};

/**
 * Why `verifyHttpRequestSignatureDetailed` rejected a request. When the request carries several
 * signatures, the reason is the one of the first signature that has `tag="agent-did"` (else the
 * first signature).
 *
 * - `missing_headers`: `Signature`, `Signature-Input` or `Signature-Agent` is absent.
 * - `malformed_headers`: `Signature` / `Signature-Input` do not parse, or no signature has a matching label.
 * - `digest_mismatch`: `Content-Digest` is absent or is not exactly the SHA-256 of the body.
 * - `tag_mismatch`: the `tag` parameter is not `agent-did` (for example another profile's signature).
 * - `alg_unsupported`: the `alg` parameter is not `ed25519`.
 * - `params_missing`: `created`, `expires`, `nonce` or `keyid` is absent or has the wrong type.
 * - `keyid_not_bound`: `keyid` is not a key of the `Signature-Agent` DID.
 * - `components_missing`: a required covered component is absent.
 * - `components_order`: the required components are all covered but not first and in order.
 * - `created_skew`: `created` is further from now than `maxCreatedSkewSeconds`.
 * - `expires_before_created`: `expires` is earlier than `created`.
 * - `lifetime_exceeded`: `expires - created` is greater than `maxSignatureLifetimeSeconds`.
 * - `expired`: now is after `expires`.
 * - `signed_component_missing`: a covered component has no value in the request (for example no `Date`).
 * - `did_resolution_failed`: the `Signature-Agent` DID has no active verification chain: unknown,
 *   revoked, deactivated, inactive controller, or the resolver failed. The SDK does not tell a
 *   resolver outage from a revocation; treat this reason as "could not establish the key".
 * - `keyid_not_authorized`: `keyid` is not an `assertionMethod` key of the DID document (a fragment
 *   the document does not have, or a key listed only under another relationship).
 * - `signature_invalid`: the signature does not verify, or the key is no longer active.
 * - `nonce_replayed`: the `(keyid, nonce)` pair was already seen.
 */
export type HttpSignatureRejection = 'missing_headers' | 'malformed_headers' | 'digest_mismatch' | 'tag_mismatch' | 'alg_unsupported' | 'params_missing' | 'keyid_not_bound' | 'components_missing' | 'components_order' | 'created_skew' | 'expires_before_created' | 'lifetime_exceeded' | 'expired' | 'signed_component_missing' | 'did_resolution_failed' | 'keyid_not_authorized' | 'signature_invalid' | 'nonce_replayed';

    ok: true;
} | {
    ok: false;
    reason: HttpSignatureRejection;

/**
 * Why a key was not authorized at the stated time (RFC-001 §6.2.2):
 * - `invalid_stated_time`: the stated time is not an RFC 3339 date-time.
 * - `did_not_yet_created`: the stated time is before the first entry of the DID log.
 * - `did_deactivated`: the DID was already deactivated at the stated time.
 * - `key_not_found`: the DID document at the stated time has no such key (or it has no public key).
 * - `key_deactivated`: the key was already deactivated at the stated time.
 * - `key_purpose_violation`: the key was not listed under the required relationship at the stated time.
 */
export type KeyAtTimeRejection = 'invalid_stated_time' | 'did_not_yet_created' | 'did_deactivated' | 'key_not_found' | 'key_deactivated' | 'key_purpose_violation';
/**
 * Why a key check at a stated time could not be decided (RFC-001 §6.2.2). The stated time is asserted by the
 * signer, so it cannot place a signature on either side of a later change:
 * - `key_retired_after_stated_time`: the key was authorized at the stated time, but a later log entry retires
 *   it (removed, deactivated or moved out of the required relationship) and no trusted time evidence places
 *   the signature before that entry.
 * - `did_deactivated_after_stated_time`: same, for a later deactivation of the DID.
 * - `history_unavailable`: no verified did:webvh log is available for the DID, so its state at the stated
 *   time cannot be established.
 */
export type KeyAtTimeUncertainty = 'key_retired_after_stated_time' | 'did_deactivated_after_stated_time' | 'history_unavailable';
/**
