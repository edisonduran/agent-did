## core/nonce-store.d.ts
/**
 * Replay protection for verified HTTP message signatures.
 *
 * `AgentIdentity.verifyHttpRequestSignature` records each accepted signature's
 * `nonce` (scoped to its `keyid`) until the signature's `expires` time and
 * rejects a second request carrying the same pair. Use a shared store (for
 * example Redis `SET key 1 NX EXAT expiresAt`) when several verifier instances
 * serve the same agents.
 *
 * Python parity: `agent_did_sdk.core.nonce_store`.
 */
export interface NonceStore {
    /**
     * Records `key` until `expiresAt` (Unix seconds). Resolves `true` when the key
     * was not recorded yet, `false` when it was (a replay). Must be atomic.
     */
    checkAndStore(key: string, expiresAt: number): Promise<boolean>;
}
export interface InMemoryNonceStoreOptions {
    /**
     * Upper bound on live entries. When the store is full after purging expired
     * entries, new nonces are refused (fail closed) instead of evicting live ones,
     * which would reopen a replay window. Default: 100 000.
     */
    maxEntries?: number;
    /** Clock in Unix seconds; injectable for tests. */
    now?: () => number;
}
/** Process-local `NonceStore`. Suitable for a single verifier instance. */
export declare class InMemoryNonceStore implements NonceStore {
    private readonly entries;
    private readonly maxEntries;
    private readonly now;
    constructor(options?: InMemoryNonceStoreOptions);
    checkAndStore(key: string, expiresAt: number): Promise<boolean>;
    /** Number of live entries (expired ones may linger until the next purge). */
    get size(): number;
    private purge;
}

/**
 * Generates a deterministic SHA-256 hash of a string payload (UTF-8 encoded).
 * Used to commit to metadata such as system prompts while allowing verification.
 *
 * @param payload The raw string to hash (e.g., "You are a helpful assistant...")
 * @returns A `0x`-prefixed lowercase hex SHA-256 digest
 */
export declare function hashPayload(payload: string): string;
/**
 * Formats a raw hash into a URI format suitable for the Agent-DID JSON-LD document.
 * Currently supports a simple hash URI scheme, but can be extended to IPFS CIDs.
 *
 * @param hashHex The raw hex hash (e.g., "0x1234...")
 * @returns A formatted URI string (e.g., "hash://sha256/1234...")
 */
export declare function formatHashUri(hashHex: string): string;
/**
 * Convenience function that hashes a payload and formats it as a URI in one step.
 *
 * @param payload The raw string to hash
 * @returns A formatted URI string
 */
export declare function generateAgentMetadataHash(payload: string): string;
/**
 * Returns the RFC 8785 (JCS) canonical JSON serialization of `value`.
 * Values are hashed exactly as given: no field (timestamps included) is
 * rewritten before canonicalization.
 *
 * @throws TypeError for non-finite numbers, bigint, or a top-level value with no JSON representation
 */
export declare function canonicalizeJson(value: unknown): string;
/**
 * Returns the `hash://sha256/<hex>` reference of a DID document: SHA-256 over the
 * UTF-8 bytes of its RFC 8785 (JCS) canonical form.
 */
export declare function generateCanonicalDocumentHash(document: unknown): string;
/**
 * Generates a random 32-byte salt (64 lowercase hex characters) for a metadata commitment.
 * The controller keeps the salt private and reveals it only to parties allowed to verify
 * the committed value.
 */
export declare function generateCommitmentSalt(): string;
/**
 * Commits to a metadata value (system prompt, model identifier) without making it guessable:
 * `hash://hmac-sha256/<hex(HMAC-SHA256(key = salt, message = UTF-8(value)))>`.
 *
 * @param value The raw value to commit to. Must not be empty.
 * @param saltHex A 32-byte salt as 64 hex characters, e.g. from `generateCommitmentSalt()`.
 */
export declare function createMetadataCommitment(value: string, saltHex: string): string;

 * Checks a revealed value (and salt) against a metadata commitment URI.

 * - `hash://hmac-sha256/...` requires the salt used at commitment time.
 * - `hash://sha256/...` (unsalted, produced by `1.0.0-rc.2` and earlier) ignores the salt.
 *

/**
 * Encodes a raw Ed25519 public key into W3C-compliant publicKeyMultibase format.
 *
 * Format: 'z' + Base58btc(multicodec_prefix + raw_public_key)
 *   - 'z' is the Multibase prefix for Base58btc
 *   - multicodec_prefix is 0xed01 for Ed25519 public keys
 *
 * @param rawPublicKey - 32-byte Ed25519 public key

 * Decodes a publicKeyMultibase string back into a raw Ed25519 public key.
 *
 * Accepts both:
 *   - Standard format: 'z' + Base58btc(0xed01 + raw_key)  (W3C compliant)

 * @param encoded - The publicKeyMultibase string
 * @returns 32-byte raw Ed25519 public key
 */
export declare function decodePublicKeyMultibase(encoded: string): Uint8Array;

export interface CreateDidWebvhLogParams {
    /** Lowercase ASCII fully qualified host name, optionally with `:port`. IP literals are rejected. */
    domain: string;
    pathSegments?: string[];
    /** Initial DID document. Every occurrence of `{DID}` is replaced with the generated DID. */
    document: Record<string, unknown>;
    /** Controller signer; its key becomes the only entry in `updateKeys`. */
    signer: AgentSigner;
    /** Optional `versionTime` of the genesis entry (UTC, second precision). Defaults to now. */
    created?: string;
}
export interface UpdateDidWebvhLogParams {
    log: string;
    signer: AgentSigner;
    /** Complete next DID document. When omitted, the current document is carried over. */
    document?: Record<string, unknown>;
    /** Replacement `updateKeys` (Ed25519 multikeys), e.g. for controller key rotation. */
    updateKeys?: string[];
    /** Optional `versionTime` for the new entry (UTC, second precision). Defaults to now. */
    updated?: string;
}
export interface DeactivateDidWebvhLogParams {
    log: string;
    signer: AgentSigner;
}
export interface ResolveDidWebvhLogOptions {
    /** When set, the log must belong to this DID. */
    did?: string;
    versionId?: string;
    versionNumber?: number;
    versionTime?: string;
}

export declare function deactivateDidWebvhLog(params: DeactivateDidWebvhLogParams): Promise<DidWebvhLogResult>;
/**
 * Verifies a `did.jsonl` log (SCID, entry hash chain, `updateKeys` proofs,
 * version times) and returns the resolved document. Throws `DidWebvhError` on
 * any verification failure; it never returns a partially verified result.
 */
export declare function resolveDidWebvhLog(logText: string, options?: ResolveDidWebvhLogOptions): Promise<DidWebvhLogResult>;
/**
 * Derives the HTTPS URL of a did:webvh DID's `did.jsonl` (did:webvh v1.0,
 * "The DID to HTTPS Transformation"). Applies the same host and path rules as
 * `createDidWebvhLog`: lowercase ASCII FQDN (no IP literals), optional port
 * encoded as `%3A`, unencoded path segments without `.`/`..`, and no path,
 * query or fragment in the DID. Throws `DidWebvhError('invalidDid')` otherwise.
 */
export declare function deriveDidWebvhLogUrl(did: string): string;
